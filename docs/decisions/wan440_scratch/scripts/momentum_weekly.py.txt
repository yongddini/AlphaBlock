def month_starts(a,b):
    d=dt.datetime.fromtimestamp(a/1000,dt.UTC).replace(day=1,hour=0,minute=0,second=0,microsecond=0)
    out=[]
    while d.timestamp()*1000<b:
        out.append(int(d.timestamp()*1000)); d=(d.replace(day=28)+dt.timedelta(days=4)).replace(day=1)
    return out
WEEK=7*DAY
def week_starts(a,b):
    # 1970-01-01은 목요일 — 월요일 0시 UTC 격자는 (t - 4일) % 7일 == 0
    t=a-((a-4*DAY)%WEEK)
    out=[]
    while t<b:
        out.append(t); t+=WEEK
    return out
def members(daily_map, a, b, look, top=True, starts=None):
    """월초 t: 종가가 t 이전(= open_time + 1일 ≤ t)인 봉만 · 직전 look일 수익률 · 상위/하위 N."""
    res={}
    for t in (starts or month_starts)(a,b):
        sc=[]
        for s,(ot,cl) in daily_map.items():
            j=bisect.bisect_right(ot, t-DAY)-1  # 마지막 확정 일봉
            if j<0: continue
            i=bisect.bisect_right(ot, ot[j]-look*DAY)-1
            if i<0 or ot[j]-ot[0] < 90*DAY: continue
            sc.append((cl[j]/cl[i]-1, s))
        sc.sort(reverse=top)
        res[t]={s for _,s in sc[:N]}
    return res
