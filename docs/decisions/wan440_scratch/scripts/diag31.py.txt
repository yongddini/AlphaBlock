g=cv[(cv.variant=='+15m')&(cv.n==31)]
def day(ms_): return dt.datetime.fromtimestamp(ms_/1000,dt.UTC).strftime('%Y-%m-%d')
for name,d in (('최악 draw5',5),('최선 draw2',2)):
    row=g[g.draw==d].iloc[0]; chosen={x+'/USDT:USDT' for x in row.symbols.split()}
    s=c.place_arm(u.restrict(spot,chosen),c.ARM_B,'bar_low',in_book=True); f=c.place_arm(u.restrict(fut,chosen),c.ARM_B,'bar_low',in_book=True)
    for label,risk in (('자기크기',row.risk),('고정3%',0.03)):
        tot,dd=c.chain(s,f,risk); fr=[t for t in f if not t.back]; bk=[t for t in f if t.back]
        ss,sf,sb=(w.simulate(x,risk=risk) for x in (s,fr,bk))
        print(f'== {name} {label} {risk:.2%}: 8.6y {tot:+.0%} MDD {dd:.1%} | 창밖 {ss.total_return:+.0%}/{ss.mdd_low:.1%} worst {day(ss.mdd_trough_ms)} | 앞 {sf.total_return:+.0%}/{sf.mdd_low:.1%} trough {day(sf.mdd_trough_ms)} | 뒤 {sb.total_return:+.0%}/{sb.mdd_low:.1%}',flush=True)
    allt=[*s,*f]
    df=pd.DataFrame({'sym':[t.symbol.split('/')[0] for t in allt],'r':[t.net_r*t.size for t in allt],'y':[day(t.exit_time)[:4] for t in allt],'d':[day(t.exit_time) for t in allt]})
    print(' 연도별 net R(크기 반영):', df.groupby('y').r.sum().round(1).to_dict())
    print(' 최악 5일:', df.groupby('d').r.sum().nsmallest(5).round(1).to_dict())
    print(' 종목별 최악 5:', df.groupby('sym').r.sum().nsmallest(5).round(1).to_dict(), '최선 5:', df.groupby('sym').r.sum().nlargest(5).round(1).to_dict())
    print(' 거래당 net R', round(df.r.mean(),4), '거래', len(df), flush=True)
    print(' 종목:', row.symbols)
