import numpy as np, pandas as pd, time, bisect
from backtest import harness
from backtest.run import parse_date_ms
from data.storage import OhlcvStore
S="/private/tmp/claude-501/-Users-joyongjin-AlphaBlock/d99ca3be-79b0-4cda-9ce5-de8a898e470d/scratchpad"
M=60000; H=60; D=1440
start=parse_date_ms(harness.DEFAULT_START)//M*M - 31*86400000; end=parse_date_ms(harness.DEFAULT_END); N=(end-start)//M+1
store=OhlcvStore(harness.DB_PATH); t0=time.time()
def load(sym):
    m=store.load(sym,"1m",start_ms=start,end_ms=end)
    if "closed" in m.columns: m=m[m["closed"].astype(bool)]
    g=(m.open_time.to_numpy(np.int64)-start)//M
    out={}
    for c in ("close","high","low","volume"):
        a=np.full(N,np.nan); a[g]=m[c].to_numpy(float); out[c]=a
    for c in ("close","high","low"): out[c]=pd.Series(out[c]).ffill().to_numpy()
    out["volume"]=np.nan_to_num(out["volume"])
    out["cv"]=np.concatenate([[0],np.cumsum(out["volume"])])
    return out
btc=load("BTC/USDT:USDT")
btc_lr=np.diff(np.log(btc["close"]),prepend=np.nan)
btc_rv=pd.Series(btc_lr).rolling(D).std().to_numpy(); btc_rv30=pd.Series(btc_lr).rolling(30*D,min_periods=10*D).std().to_numpy()
for k in (1,2):
    t=pd.read_csv(f"{S}/bw_trades_k{k}.csv")
    ref=pd.read_csv("backtest/reports/wan434_trades.csv.gz"); ref=ref[ref["조합"]=="ts4·하한4%·%K<25"][["종목","시간봉","진입시각_ms","진입시%K","손절폭(%)"]]
    ref.columns=["sym","tf","e","kval","sw"]
    t=t.merge(ref.drop_duplicates(["sym","tf","e"]),on=["sym","tf","e"],how="left")
    rows=[]
    for sym,g in t.groupby("sym"):
        a=load(sym)
        for i,r in g.iterrows():
            j=(int(r.e)-start)//M - 1   # 진입 직전 확정 1분봉
            c=a["close"]; ret=lambda n,arr=c: arr[j]/arr[j-n]-1
            f=dict(idx=i)
            for n,nm in ((H,"1h"),(4*H,"4h"),(D,"24h"),(7*D,"7d")):
                f[f"ret_{nm}"]=ret(n); 
                if nm!="7d": f[f"btc_{nm}"]=btc["close"][j]/btc["close"][j-n]-1; f[f"rel_{nm}"]=f[f"ret_{nm}"]-f[f"btc_{nm}"]
            f["dd24h"]=c[j]/np.nanmax(a["high"][j-D:j+1])-1
            v1h=a["cv"][j+1]-a["cv"][j+1-H]; v7=(a["cv"][j+1]-a["cv"][j+1-7*D])/(7*24)
            f["vol_ratio"]=v1h/v7 if v7>0 else np.nan
            lo15=np.nanmin(a["low"][j-15:j+1]); f["bounce15"]=c[j]/lo15-1
            f["btc_vol_ratio"]=btc_rv[j]/btc_rv30[j] if btc_rv30[j]>0 else np.nan
            f["hour"]=((int(r.e)//3600000)%24)
            rows.append(f)
        print(k,sym,f"{time.time()-t0:.0f}s",flush=True)
    F=pd.DataFrame(rows).set_index("idx"); t=t.join(F)
    e_all=np.sort(t.e.to_numpy()); t["prior24"]=[bisect.bisect_left(e_all,v)-bisect.bisect_left(e_all,v-86400000) for v in t.e]
    t.to_csv(f"{S}/feat_k{k}.csv",index=False)
