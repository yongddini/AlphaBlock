"""후보 풀 확대(%K 조건 없음 · 하한 FLOOR) → 시간청산 ts4 · 손절 k배 → 채택 북 배치 → 거래 기록 + 진입 특징."""
import dataclasses, sys, time
import numpy as np, pandas as pd
from pathlib import Path
from backtest import harness
from backtest.book_cli import net_r
from backtest.models import ExitReason
from backtest.run import parse_date_ms
from backtest.wan424_stoch_ob_arm import build_base_payloads, arm_pool, stoch_series, k_before, place_segments
from common.costs import Liquidity
from data.models import timeframe_to_ms
from data.storage import OhlcvStore
S="/private/tmp/claude-501/-Users-joyongjin-AlphaBlock/d99ca3be-79b0-4cda-9ce5-de8a898e470d/scratchpad"
FLOOR=float(sys.argv[1]); KS=(1,2); HOLD=4; M=60000
t0=time.time()
payloads=build_base_payloads(jobs=4,payload_dir=Path("backtest/cache/wan424_payloads"))
store=OhlcvStore(harness.DB_PATH)
start,end=parse_date_ms(harness.DEFAULT_START),parse_date_ms(harness.DEFAULT_END)
fstart=start-8*86400000
by={}
for p in payloads: by.setdefault(p.symbol,[]).append(p)
new={k:{} for k in KS}; feat={}
for sym,ps in by.items():
    m=store.load(sym,"1m",start_ms=fstart,end_ms=end)
    if "closed" in m.columns: m=m[m["closed"].astype(bool)]
    m=m.sort_values("open_time"); ot=m.open_time.to_numpy(np.int64); lo=m.low.to_numpy(float); cl=m.close.to_numpy(float); hi=m.high.to_numpy(float)
    for p in ps:
        tf=timeframe_to_ms(p.timeframe); kt,kv=stoch_series(store,p.symbol,p.timeframe)
        res={k:[] for k in KS}
        for c in arm_pool(p.candidates.get("full",()),min_width=FLOOR):
            j=int(np.searchsorted(ot,c.entry_time))
            if j>=len(ot) or ot[j]!=c.entry_time or j<7*1440: continue
            pj=j-1
            def back(n):
                i=np.searchsorted(ot,ot[pj]-n*M,side="left"); return cl[pj]/cl[i]-1
            feat[(p.symbol,p.timeframe,int(c.entry_time))]=dict(ret_4h=back(240),ret_24h=back(1440),
                dd24h=cl[pj]/hi[np.searchsorted(ot,ot[pj]-1440*M):pj+1].max()-1,kval=k_before(kt,kv,int(c.entry_time),tf))
            b=ot[j:]//tf; pos=np.flatnonzero(np.diff(b)!=0); je=j+int(pos[HOLD-1])+1 if len(pos)>=HOLD else len(ot)
            e=float(c.entry_price); d=e-float(c.stop_price)
            for k in KS:
                stop=e-k*d; hit=np.flatnonzero(lo[j:je]<=stop)
                ix,px,rs=(j+int(hit[0]),stop,ExitReason.STOP_LOSS) if len(hit) else (je-1,float(cl[je-1]),ExitReason.END_OF_DATA)
                res[k].append(dataclasses.replace(c,stop_price=stop,entry_time=int(ot[j]),exit_time=int(ot[ix]),exit_price=px,reason=rs,
                    take_profit_price=None,entry_liquidity=Liquidity.MAKER,mfe_r=None,mae_r=None))
        for k in KS: new[k][(p.symbol,p.timeframe)]=tuple(res[k])
    print(sym,f"{time.time()-t0:.0f}s",flush=True)
F=pd.DataFrame([dict(sym=a,tf=b,e=c,**v) for (a,b,c),v in feat.items()])
F.to_csv(f"{S}/pool{FLOOR}_feat.csv",index=False)
for k in KS:
    print(f"k={k} 후보 {sum(len(v) for v in new[k].values())}",flush=True)
    pl=[dataclasses.replace(p,candidates={"full":new[k][(p.symbol,p.timeframe)],"is":(),"oos":()},reentry_candidates={},arm_candidates={}) for p in payloads]
    bnd={(p.symbol,p.timeframe):p.boundary_ms for p in payloads}
    seg=next(s for s in place_segments(pl) if s.segment=="full")
    tr=[(t.exit_time,t.entry_time,net_r(t,pp),t.exits[-1].reason==ExitReason.STOP_LOSS,pp.trigger_time>=bnd[pp.cell],pp.cell[0],pp.cell[1],float(t.entry_price),float(pp.stop_price)) for t,pp in seg.trades_with_placements()]
    pd.DataFrame(tr,columns=["x","e","r","stop","back","sym","tf","ep","sp"]).to_csv(f"{S}/pool{FLOOR}_k{k}.csv",index=False)
    print(f"k={k} 배치 {len(tr)}",flush=True)
print("done",f"{time.time()-t0:.0f}s")
