import numpy as np, pandas as pd, time
from backtest import harness
from backtest.run import parse_date_ms
from data.storage import OhlcvStore
S="/private/tmp/claude-501/-Users-joyongjin-AlphaBlock/d99ca3be-79b0-4cda-9ce5-de8a898e470d/scratchpad"
M=60000; D=86400000
start=parse_date_ms(harness.DEFAULT_START)//M*M; end=parse_date_ms(harness.DEFAULT_END)
N=(end-start)//M+1
store=OhlcvStore(harness.DB_PATH)
T={k:pd.read_csv(f"{S}/bw_trades_k{k}.csv") for k in (1,2,4,6,8)}
syms=sorted(set().union(*[set(t.sym) for t in T.values()]))
# 1) 복리 사이징: 진입 순간 확정 자본의 1% (이벤트 순서: 같은 시각은 청산 먼저 — 엔진 규약)
W={}; R={}
for k,t in T.items():
    ev=sorted([(x,0,i) for i,x in enumerate(t.x)]+[(e,1,i) for i,e in enumerate(t.e)])
    eq=1.0; w=np.zeros(len(t))
    for tm,kind,i in ev:
        if kind==1: w[i]=eq
        else: eq+=0.01*w[i]*t.r.iat[i]
    W[k]=w
    r=np.zeros(N); np.add.at(r,((t.x.to_numpy()-start)//M).astype(np.int64),0.01*w*t.r.to_numpy()); R[k]=1+np.cumsum(r)
U={k:np.zeros(N) for k in T}
t0=time.time()
for sym in syms:
    m=store.load(sym,"1m",start_ms=start,end_ms=end)
    if "closed" in m.columns: m=m[m["closed"].astype(bool)]
    g=((m.open_time.to_numpy(np.int64)-start)//M)
    low=np.full(N,np.nan); low[g]=m.low.to_numpy(float)
    low=pd.Series(low).ffill().to_numpy()
    for k,t in T.items():
        s=t[t.sym==sym]
        for i,row in s.iterrows():
            a=(int(row.e)-start)//M; b=(int(row.x)-start)//M
            if b<=a: continue
            U[k][a:b]+=0.01*W[k][i]*(low[a:b]-row.ep)/(row.ep-row.sp)
print(f"평가 {time.time()-t0:.0f}s")
day=(np.arange(N)*M+start)//D
for k in T:
    mtm=R[k]+U[k]; rl=R[k]
    pk=np.maximum.accumulate(mtm); mdd=(1-mtm/pk).max(); i=int(np.argmax(1-mtm/pk)); j=int(np.argmax(mtm[:i+1]))
    rmdd=(1-rl/np.maximum.accumulate(rl)).max()
    df=pd.DataFrame({"d":day,"v":mtm}); g=df.groupby("d").v.agg(["first","min"])
    prev_close=df.groupby("d").v.last().shift(1).fillna(1.0)
    drop=(g["min"]/prev_close-1)
    wd=drop.nsmallest(3)
    ts=lambda x: pd.to_datetime(x*M+start,unit="ms").strftime("%Y-%m-%d %H:%M")
    print(f"{k}배 | 최종 {mtm[-1]-1:+.0%} · 평가손 포함 MDD {mdd:.1%} (확정만 {rmdd:.1%}) · 고점 {ts(j)} → 바닥 {ts(i)} | 하루 최악(전일 마감 대비 장중 최저) "
          + ", ".join(f"{pd.to_datetime(d*D,unit='ms').strftime('%Y-%m-%d')} {v:+.1%}" for d,v in wd.items())
          + f" | −10% 이하인 날 {(drop<=-0.10).sum()} · −20% 이하 {(drop<=-0.20).sum()}")
