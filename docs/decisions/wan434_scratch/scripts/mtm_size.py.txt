def simulate(k, rule, param=None, f=0.01, throttle=False, sel=None, cap=None, brk=None):
    t=T[k]; order=np.argsort(t.e.to_numpy(),kind="stable")
    ev=sorted([(int(t.x.iat[i]),0,i) for i in range(len(t))]+[(int(t.e.iat[i]),1,i) for i in range(len(t))])
    e_all=np.sort(t.e.to_numpy())  # 신호 수 = 기준 거래(체결 여부 무관)
    eq=1.0; peak=1.0; w=np.zeros(len(t)); keep=np.zeros(len(t),bool); open_n=0; open_risk=0.0; stops=[]
    for tm,kind,i in ev:
        if kind==0:
            if keep[i]: eq+=f*w[i]*t.r.iat[i]; open_n-=1; peak=max(peak,eq); open_risk-=f*w[i]
            if keep[i] and t.r.iat[i]<-0.5: stops.append(tm)
            continue
        size=1.0
        if sel is not None and not sel[i]: continue
        ps=bisect.bisect_left(e_all,tm)-bisect.bisect_left(e_all,tm-D)
        if rule=="cap" and open_n>=param: continue
        if rule in ("skip","skip+budget") and 1<=ps<=10: continue
        if rule in ("budget","skip+budget"): size=min(1.0,10.0/(ps+1))
        if throttle: size*=min(1.0,max(0.2,1-(1-eq/peak)/0.30))
        if brk is not None:
            K,win=brk; lo_=tm-win*60000
            if sum(1 for x in stops[-K*4:] if x>lo_)>=K: continue
        if cap is not None and (open_risk+f*eq*size)/eq>cap: continue
        keep[i]=True; w[i]=eq*size; open_n+=1; open_risk+=f*w[i]
    R=np.zeros(N); U=np.zeros(N)
    for i in np.flatnonzero(keep):
        a,b,path=P[k][i]; R[(int(t.x.iat[i])-start)//M]+=f*w[i]*t.r.iat[i]
        if b>a: U[a:b]+=f*w[i]*path
    mtm=1+np.cumsum(R)+U
    pk=np.maximum.accumulate(mtm); dd=1-mtm/pk; mdd=dd.max(); global LAST; ii=int(np.argmax(dd)); LAST=(int(np.argmax(mtm[:ii+1])),ii)
    day=(np.arange(N)*M+start)//D; df=pd.DataFrame({"d":day,"v":mtm})
    prev=df.groupby("d").v.last().shift(1).fillna(1.0); worst=(df.groupby("d").v.min()/prev-1).min()
    yrs=(end-start)/D/365.25
    return keep.sum(), mtm[-1]-1, (mtm[-1])**(1/yrs)-1, mdd, worst
TARGET=0.30
def at_target(k,rule,sel,cap=None,brk=None):
    lo,hi=0.0005,0.2; best=None
    for _ in range(14):

        if r[3]<=TARGET: lo=f; best=(f,)+r
        else: hi=f
    return best
