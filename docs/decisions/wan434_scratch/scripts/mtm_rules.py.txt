import numpy as np, pandas as pd, time, bisect
from backtest import harness
from backtest.run import parse_date_ms
from data.storage import OhlcvStore
S="/private/tmp/claude-501/-Users-joyongjin-AlphaBlock/d99ca3be-79b0-4cda-9ce5-de8a898e470d/scratchpad"
M=60000; D=86400000; start=parse_date_ms(harness.DEFAULT_START)//M*M; end=parse_date_ms(harness.DEFAULT_END); N=(end-start)//M+1
KS=(1,4,8); T={k:pd.read_csv(f"{S}/bw_trades_k{k}.csv").reset_index(drop=True) for k in KS}
store=OhlcvStore(harness.DB_PATH); t0=time.time()
# 거래별 미실현 R 경로(1분 저가) 미리 계산
P={k:[None]*len(T[k]) for k in KS}
for sym in sorted(set().union(*[set(t.sym) for t in T.values()])):
    m=store.load(sym,"1m",start_ms=start,end_ms=end)
    if "closed" in m.columns: m=m[m["closed"].astype(bool)]
    low=np.full(N,np.nan); low[(m.open_time.to_numpy(np.int64)-start)//M]=m.low.to_numpy(float); low=pd.Series(low).ffill().to_numpy()
    for k,t in T.items():
        for i in np.flatnonzero((t.sym==sym).to_numpy()):
            r=t.iloc[i]; a=(int(r.e)-start)//M; b=(int(r.x)-start)//M
            P[k][i]=(a,b,(low[a:b]-r.ep)/(r.ep-r.sp) if b>a else np.zeros(0))
print(f"경로 준비 {time.time()-t0:.0f}s",flush=True)
def simulate(k, rule, param=None):
    t=T[k]; order=np.argsort(t.e.to_numpy(),kind="stable")
    ev=sorted([(int(t.x.iat[i]),0,i) for i in range(len(t))]+[(int(t.e.iat[i]),1,i) for i in range(len(t))])
    e_all=np.sort(t.e.to_numpy())  # 신호 수 = 기준 거래(체결 여부 무관)
    eq=1.0; w=np.zeros(len(t)); keep=np.zeros(len(t),bool); open_n=0
    for tm,kind,i in ev:
        if kind==0:
            if keep[i]: eq+=0.01*w[i]*t.r.iat[i]; open_n-=1
            continue
        size=1.0
        ps=bisect.bisect_left(e_all,tm)-bisect.bisect_left(e_all,tm-D)
        if rule=="cap" and open_n>=param: continue
        if rule in ("skip","skip+budget") and 1<=ps<=10: continue
        if rule in ("budget","skip+budget"): size=min(1.0,10.0/(ps+1))
        keep[i]=True; w[i]=eq*size; open_n+=1
    R=np.zeros(N); U=np.zeros(N)
    for i in np.flatnonzero(keep):
        a,b,path=P[k][i]; R[(int(t.x.iat[i])-start)//M]+=0.01*w[i]*t.r.iat[i]
        if b>a: U[a:b]+=0.01*w[i]*path
    mtm=1+np.cumsum(R)+U
    pk=np.maximum.accumulate(mtm); mdd=(1-mtm/pk).max()
    day=(np.arange(N)*M+start)//D; df=pd.DataFrame({"d":day,"v":mtm})
    prev=df.groupby("d").v.last().shift(1).fillna(1.0); worst=(df.groupby("d").v.min()/prev-1).min()
    yrs=(end-start)/D/365.25
    return keep.sum(), mtm[-1]-1, (mtm[-1])**(1/yrs)-1, mdd, worst
rows=[]
for k in KS:
    for rule,param in [("none",None),("cap",5),("cap",10),("cap",15),("skip",None),("budget",None),("skip+budget",None)]:
        n,ret,cagr,mdd,worst=simulate(k,rule,param)
        name={"none":"규칙 없음","cap":f"동시 보유 ≤{param}","skip":"초반 건너뛰기","budget":"24h 예산 10%","skip+budget":"건너뛰기+예산"}[rule]
        rows.append((k,name,n,ret,cagr,mdd,worst,ret/mdd))
        print(f"{k}배 | {name:12s} | 거래 {n} | 6년 {ret:+.0%} · 연 {cagr:+.1%} | 평가손 포함 MDD {mdd:.1%} | 하루 최악 {worst:+.1%} | 수익/MDD {ret/mdd:.1f}",flush=True)
pd.DataFrame(rows,columns=["k","rule","n","ret","cagr","mdd","worst","ratio"]).to_csv(f"{S}/mtm_rules.csv",index=False)
