import pandas as pd, numpy as np, bisect
from backtest import harness
from backtest.run import parse_date_ms
from data.storage import OhlcvStore
S="/private/tmp/claude-501/-Users-joyongjin-AlphaBlock/d99ca3be-79b0-4cda-9ce5-de8a898e470d/scratchpad"
M=60000; D=86400000; H=3600000
F=pd.read_csv(f"{S}/feat_k2.csv")
e_all=np.sort(F.e.to_numpy()); stops_x=np.sort(F.x[F.stop].to_numpy())
cnt=lambda arr,v,w: bisect.bisect_left(arr,v)-bisect.bisect_left(arr,v-w)
F["ps"]=[cnt(e_all,v,D) for v in F.e]; F["sig1h"]=[cnt(e_all,v,H) for v in F.e]; F["sig4h"]=[cnt(e_all,v,4*H) for v in F.e]
F["stops1h"]=[bisect.bisect_right(stops_x,v)-bisect.bisect_right(stops_x,v-H) for v in F.e]   # 진입 순간까지 확정된 손절(청산 ≤ 진입)
F["stops4h"]=[bisect.bisect_right(stops_x,v)-bisect.bisect_right(stops_x,v-4*H) for v in F.e]
start=int(F.e.min())-2*D; end=int(F.e.max())
st=OhlcvStore(harness.DB_PATH); m=st.load("BTC/USDT:USDT","1m",start_ms=start,end_ms=end)
m=m[m["closed"].astype(bool)].sort_values("open_time"); ot=m.open_time.to_numpy(np.int64); cl=m.close.to_numpy(float); lo=m.low.to_numpy(float)
def at(v): return int(np.searchsorted(ot,v))-1
F["btc_15m"]=[cl[at(v)]/cl[at(v-15*M)]-1 for v in F.e]
F["btc_from_low24"]=[cl[at(v)]/lo[at(v-D):at(v)+1].min()-1 for v in F.e]
F["btc_from_low1h"]=[cl[at(v)]/lo[at(v-H):at(v)+1].min()-1 for v in F.e]
F.to_csv(f"{S}/feat_k2_plus.csv",index=False)
