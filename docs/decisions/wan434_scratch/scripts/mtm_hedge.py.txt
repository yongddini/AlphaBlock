import numpy as np, pandas as pd, time, bisect
from backtest import harness
from backtest.run import parse_date_ms
from data.storage import OhlcvStore
S="/private/tmp/claude-501/-Users-joyongjin-AlphaBlock/d99ca3be-79b0-4cda-9ce5-de8a898e470d/scratchpad"
M=60000; D=86400000; start=parse_date_ms(harness.DEFAULT_START)//M*M; end=parse_date_ms(harness.DEFAULT_END); N=(end-start)//M+1
KS=(1,2); T={k:pd.read_csv(f"{S}/bw_trades_k{k}.csv").reset_index(drop=True) for k in KS}
store=OhlcvStore(harness.DB_PATH); t0=time.time()
def arr(sym,col):
    m=store.load(sym,"1m",start_ms=start,end_ms=end)
    if "closed" in m.columns: m=m[m["closed"].astype(bool)]
    a=np.full(N,np.nan); a[(m.open_time.to_numpy(np.int64)-start)//M]=m[col].to_numpy(float); return pd.Series(a).ffill().bfill().to_numpy()
BTC="BTC/USDT:USDT"; btc=arr(BTC,"close")
P={k:[None]*len(T[k]) for k in KS}
for sym in sorted(set().union(*[set(t.sym) for t in T.values()])):
    low=arr(sym,"low")
    for k,t in T.items():
        for i in np.flatnonzero((t.sym==sym).to_numpy()):
            r=t.iloc[i]; a=(int(r.e)-start)//M; b=(int(r.x)-start)//M; lev=r.ep/(r.ep-r.sp)
            hp=(btc[a:b]/btc[a]-1) if b>a else np.zeros(0); hx=btc[b]/btc[a]-1
            P[k][i]=(a,b,(low[a:b]-r.ep)/(r.ep-r.sp) if b>a else np.zeros(0), lev, hp, hx, sym==BTC)
print(f"경로 준비 {time.time()-t0:.0f}s",flush=True)
HC=0.0018  # 헤지 왕복 비용(명목 대비): 테이커 4bp×2 + 슬리피지 5bp×2
def simulate(k, rule, f, beta):
    t=T[k]; e_all=np.sort(t.e.to_numpy())
    ev=sorted([(int(t.x.iat[i]),0,i) for i in range(len(t))]+[(int(t.e.iat[i]),1,i) for i in range(len(t))])
    eq=1.0; w=np.zeros(len(t)); keep=np.zeros(len(t),bool); hz=np.zeros(len(t))
    for tm,kind,i in ev:
        a,b,path,lev,hp,hx,isbtc=P[k][i]
        if kind==0:
            if keep[i]:
                hedge=0.0 if isbtc else -beta*f*w[i]*lev*(hx)-beta*f*w[i]*lev*HC
                hz[i]=hedge; eq+=f*w[i]*t.r.iat[i]+hedge
            continue
        ps=bisect.bisect_left(e_all,tm)-bisect.bisect_left(e_all,tm-D); size=1.0
        if rule=="skip+budget":
            if 1<=ps<=10: continue
            size=min(1.0,10.0/(ps+1))
        keep[i]=True; w[i]=eq*size
    R=np.zeros(N); U=np.zeros(N)
    for i in np.flatnonzero(keep):
        a,b,path,lev,hp,hx,isbtc=P[k][i]
        R[(int(t.x.iat[i])-start)//M]+=f*w[i]*t.r.iat[i]+hz[i]
        if b>a:
            U[a:b]+=f*w[i]*path
            if not isbtc: U[a:b]+=-beta*f*w[i]*lev*hp
    mtm=1+np.cumsum(R)+U; mdd=(1-mtm/np.maximum.accumulate(mtm)).max()
    df=pd.DataFrame({"d":(np.arange(N)*M+start)//D,"v":mtm}); prev=df.groupby("d").v.last().shift(1).fillna(1.0); worst=(df.groupby("d").v.min()/prev-1).min()
    yrs=(end-start)/D/365.25
    return keep.sum(), mtm[-1]-1, max(mtm[-1],1e-9)**(1/yrs)-1, mdd, worst
for k,rule in [(1,"none"),(1,"skip+budget"),(2,"skip+budget")]:
    for beta in (0.0,0.5,1.0):
        lo,hi,best=0.0005,0.2,None
        for _ in range(14):
            f=(lo*hi)**0.5; r=simulate(k,rule,f,beta)
            if r[3]<=0.30: lo=f; best=(f,)+r
            else: hi=f
        f,n,ret,cagr,mdd,worst=best
        print(f"{k}배 {rule:12s} 헤지 {beta:.1f} | 거래당 {f:.2%} | 거래 {n} | 6년 {ret:+.0%} · 연 {cagr:+.1%} | 평가손 MDD {mdd:.1%} | 하루 최악 {worst:+.1%}",flush=True)
