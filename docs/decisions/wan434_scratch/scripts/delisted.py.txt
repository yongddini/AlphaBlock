"""탐색: LUNA·FTT(붕괴한 시총 상위 코인)를 유니버스에 넣고 손절거리 k배를 채택 북에서 잰다.
데이터는 data.binance.vision 월별 1분봉 zip → 스크래치 SQLite(운영 DB 불변)."""
import dataclasses, glob, io, sys, time, zipfile
import numpy as np, pandas as pd
from pathlib import Path
from backtest import harness
from backtest import wan424_stoch_ob_arm as w424  # 3h 등록
from backtest.book_cli import net_r
from backtest.models import ExitReason
from backtest.run import parse_date_ms
from backtest.wan169_leverage_book import run_cells
from common.costs import Liquidity
from data.aggregate import build_symbol
from data.models import Candle, timeframe_to_ms
from data.storage import OhlcvStore
S = Path("/private/tmp/claude-501/-Users-joyongjin-AlphaBlock/d99ca3be-79b0-4cda-9ce5-de8a898e470d/scratchpad")
ZIP = Path("/Users/joyongjin/AlphaBlock/data/cache/wan434_delisted")
SDB = S / "delisted.db"; SCACHE = S / "delisted_cache"; SCACHE.mkdir(exist_ok=True)
NEWS = {"LUNAUSDT": "LUNA/USDT:USDT", "FTTUSDT": "FTT/USDT:USDT"}
KS = (1, 4, 8); HOLD, FLOOR, THR = 4, 0.04, 25.0
t0 = time.time()
sstore = OhlcvStore(SDB, cache_dir=SCACHE)
for raw, sym in NEWS.items():
    if sstore.count(sym, "1m"): continue
    for f in sorted(ZIP.glob(f"{raw}-1m-*.zip")):
        with zipfile.ZipFile(f) as z:
            txt = z.read(z.namelist()[0]).decode()
        df = pd.read_csv(io.StringIO(txt), header=None)
        if not str(df.iloc[0, 0]).isdigit(): df = df.iloc[1:]
        df = df.iloc[:, :6].astype(float)
        sstore.upsert_candles(Candle(sym, "1m", int(r[0]), r[1], r[2], r[3], r[4], r[5], True) for r in df.itertuples(index=False))
    build_symbol(sstore, sym, w424.TIMEFRAMES)
    print(sym, "적재", sstore.count(sym, "1m"), "1m행", f"{time.time()-t0:.0f}s", flush=True)
# 새 종목 base 후보 — 채택 북 팔과 **같은 인자**(build_base_payloads), DB만 스크래치로
kw = harness.load_market_data.__kwdefaults__
kw["db_path"], kw["cache_dir"] = str(SDB), str(SCACHE)
new_payloads = run_cells(tuple(NEWS.values()), w424.TIMEFRAMES, start=harness.DEFAULT_START, end=harness.DEFAULT_END,
    jobs=1, take_profit_liquidity=harness.ADOPTED_TAKE_PROFIT_LIQUIDITY, retap_mode="once", reentry=False,
    engine_check=False, fill=harness.fill_preset("pen_5bp"), no_same_step_tp=True)
kw["db_path"], kw["cache_dir"] = harness.DB_PATH, harness.CACHE_DIR
print("새 종목 후보", {(p.symbol, p.timeframe): len(p.candidates["full"]) for p in new_payloads}, f"{time.time()-t0:.0f}s", flush=True)
old_payloads = w424.build_base_payloads(jobs=4, payload_dir=Path("backtest/cache/wan424_payloads"))
start, end = parse_date_ms(harness.DEFAULT_START), parse_date_ms(harness.DEFAULT_END)
def arms(payloads, store):
    out = {k: {} for k in KS}; by = {}
    for p in payloads: by.setdefault(p.symbol, []).append(p)
    for sym, ps in by.items():
        m = store.load(sym, "1m", start_ms=start, end_ms=end)
        if "closed" in m.columns: m = m[m["closed"].astype(bool)]
        m = m.sort_values("open_time"); ot = m.open_time.to_numpy(np.int64); lo = m.low.to_numpy(float); cl = m.close.to_numpy(float)
        for p in ps:
            tf = timeframe_to_ms(p.timeframe); kt, kv = w424.stoch_series(store, p.symbol, p.timeframe)
            res = {k: [] for k in KS}
            for c in w424.arm_pool(p.candidates.get("full", ()), min_width=FLOOR):
                v = w424.k_before(kt, kv, int(c.entry_time), tf)
                if v is None or v >= THR: continue
                j = int(np.searchsorted(ot, c.entry_time))
                if j >= len(ot) or ot[j] != c.entry_time: continue
                b = ot[j:] // tf; pos = np.flatnonzero(np.diff(b) != 0)
                je = j + int(pos[HOLD - 1]) + 1 if len(pos) >= HOLD else len(ot)
                e = float(c.entry_price); d = e - float(c.stop_price)
                for k in KS:
                    stop = e - k * d; hit = np.flatnonzero(lo[j:je] <= stop)
                    ix, px, rs = (j + int(hit[0]), stop, ExitReason.STOP_LOSS) if len(hit) else (je - 1, float(cl[je - 1]), ExitReason.END_OF_DATA)
                    res[k].append(dataclasses.replace(c, stop_price=stop, entry_time=int(ot[j]), exit_time=int(ot[ix]), exit_price=px,
                        reason=rs, take_profit_price=None, entry_liquidity=Liquidity.MAKER, mfe_r=None, mae_r=None))
            for k in KS: out[k][(p.symbol, p.timeframe)] = tuple(res[k])
    return out
A_old = arms(old_payloads, OhlcvStore(harness.DB_PATH)); A_new = arms(new_payloads, sstore)
print("팔 후보 완료", f"{time.time()-t0:.0f}s", {k: sum(len(v) for v in A_new[k].values()) for k in KS}, flush=True)
def run(payloads, A, k, label):
    pl = [dataclasses.replace(p, candidates={"full": A[k][(p.symbol, p.timeframe)], "is": (), "oos": ()}, reentry_candidates={}, arm_candidates={}) for p in payloads]
    bnd = {(p.symbol, p.timeframe): p.boundary_ms for p in payloads}
    seg = next(s for s in w424.place_segments(pl) if s.segment == "full")
    rows = [(t.exit_time, t.entry_time, net_r(t, pp), pp.cell[0], pp.trigger_time >= bnd[pp.cell]) for t, pp in seg.trades_with_placements()]
    df = pd.DataFrame(rows, columns=["x", "e", "r", "sym", "back"]).sort_values("x", kind="stable")
    def path(r):
        fe = 1 + np.cumsum(0.01 * r); ce = np.cumprod(1 + 0.01 * r)
        return fe[-1] - 1, (1 - fe / np.maximum.accumulate(np.maximum(fe, 1))).max(), ce[-1] - 1, (1 - ce / np.maximum.accumulate(np.maximum(ce, 1))).max()
    f = path(df.r.to_numpy()); a = path(df[~df.back].r.to_numpy()); b = path(df[df.back].r.to_numpy())
    nw = df[df.sym.isin(NEWS.values())]
    print(f"{label} {k}배 | 거래 {len(df)} · {df.r.mean():+.3f}R · 고정 {f[0]:+.0%}/MDD {f[1]:.0%} (비 {f[0]/f[1]:.1f}) · 복리 {f[2]:+.0%}/MDD {f[3]:.0%} · 앞 비 {a[0]/a[1]:.1f} · 뒤 비 {b[0]/b[1]:.1f}"
          + (f" || LUNA·FTT 거래 {len(nw)} · 합 {nw.r.sum():+.1f}R · 최악 {nw.r.min():+.2f}R · LUNA {nw[nw.sym=='LUNA/USDT:USDT'].r.sum():+.1f}R · FTT {nw[nw.sym=='FTT/USDT:USDT'].r.sum():+.1f}R" if len(nw) else ""), flush=True)
    return df
for k in KS:
    base = run(old_payloads, A_old, k, "기존31")
    if k == 1: print("검산 기존31 1배 거래", len(base), "net R 합", round(base.r.sum(), 6), "(공개 2710 · 325.035733)")
    allp = old_payloads + list(new_payloads); AA = {kk: {**A_old[kk], **A_new[kk]} for kk in KS}
    df = run(allp, AA, k, "+LUNA·FTT")
    df[df.sym.isin(NEWS.values())].to_csv(S / f"delisted_trades_k{k}.csv", index=False)
print("done", f"{time.time()-t0:.0f}s")
