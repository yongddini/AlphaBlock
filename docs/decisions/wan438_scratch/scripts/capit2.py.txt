import pickle, sys, dataclasses
sys.path.insert(0, '.')
from rsim import Rules, chain, run, summary
from search import fit
from capit import scaled, T
def ev(fill, arm, mult, vol, crash):
    s, f = scaled('spot', T[('spot', fill, arm)], mult, vol, crash), scaled('fut', T[('fut', fill, arm)], mult, vol, crash)
    r = fit(lambda x: chain(run(s, risk=x), run(f, risk=x))[1]); tot = chain(run(s, risk=r), run(f, risk=r))[0]
    bk = [t for t in f if t.back]; fr=[t for t in f if not t.back]
    rs = fit(lambda x: summary(run(s, risk=x))[1]); rb = fit(lambda x: summary(run(bk, risk=x))[1]); rfr = fit(lambda x: summary(run(fr, risk=x))[1])
    return r, tot, summary(run(s, risk=rs))[0], summary(run(fr, risk=rfr))[0], summary(run(bk, risk=rb))[0]
for crash in (-0.03, -0.05, -0.08):
    for vol in (4.0, 6.0, 8.0, 12.0, 1e12):
        r, tot, sp, fr, bk = ev('stop', 'gate', 0.5, vol, crash)
        lab = '예외없음(대조군)' if vol > 1e9 else f'투매≥{vol:g}배'
        print(f"폭락 {crash:+.0%} {lab:14s} 크기 {r:.2%} | 8.6년30% {tot:+6.0%} | 창밖 {sp:+.0%} 앞 {fr:+.0%} 뒤 {bk:+.0%}", flush=True)
