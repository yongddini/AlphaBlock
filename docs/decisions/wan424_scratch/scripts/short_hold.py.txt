TFS=("1h","2h","3h","4h","6h","8h","12h","1d","1w"); HOLDS=(1,2,3,4)
def work(p):
    import tf3h_patch  # noqa
    shorts={seg:[c for c in v if c.side.sign<0 and not c.is_reentry and c.tap_index==0 and c.entry_price>0 and arm.stop_width(c)>=0.03] for seg,v in p.candidates.items()}
    if not any(shorts.values()): return (p.symbol,p.timeframe),{}
    m=harness.load_market_data(p.symbol,p.timeframe,start_ms=parse_date_ms(harness.DEFAULT_START),end_ms=parse_date_ms(harness.DEFAULT_END),need_1m=True)
    win={SEGMENT_FULL:m,SEGMENT_IS:harness.slice_market(m,IS_SEGMENT),SEGMENT_OOS:harness.slice_market(m,OOS_SEGMENT)}
    tf=timeframe_to_ms(p.timeframe); out={}
    for seg,cs in shorts.items():
        if not cs: out[seg]={h:() for h in HOLDS}; continue
        d=arm.derive_hold_arms(cs,substeps=build_substeps(win[seg].df_1m,tf),holds=HOLDS)
        out[seg]={h:tuple(d[h]) for h in HOLDS}
    return (p.symbol,p.timeframe),out
