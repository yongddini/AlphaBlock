def rsi_line(close,n=14):
    out=[None]*len(close)
    if len(close)<=n: return out
    g=[max(close[i]-close[i-1],0) for i in range(1,len(close))]; l=[max(close[i-1]-close[i],0) for i in range(1,len(close))]
    ag=sum(g[:n])/n; al=sum(l[:n])/n; out[n]=100 if al==0 else 100-100/(1+ag/al)
    for i in range(n+1,len(close)):
        ag=(ag*(n-1)+g[i-1])/n; al=(al*(n-1)+l[i-1])/n; out[i]=100 if al==0 else 100-100/(1+ag/al)
    return out

def at(m,sym,tf,ms):
    times,v=m[(sym,tf)]
    if not times: return None
    tfm=timeframe_to_ms(tf); b=(ms//tfm)*tfm; i=bisect.bisect_left(times,b)
    if i>=len(times) or times[i]!=b:
        i=bisect.bisect_right(times,b)-1
        if i<0: return None
    return v[i-1] if i-1>=0 else None
