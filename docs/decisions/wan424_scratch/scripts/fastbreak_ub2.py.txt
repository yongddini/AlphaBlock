with gzip.open("backtest/cache/wan424_iso_work.pkl.gz") as fh: cl=pickle.load(fh)["cells"]
def path(tr):
    eq=peak=1.0; mdd=0.0
    for _,r in sorted(tr): eq+=0.01*eq*r; peak=max(peak,eq); mdd=max(mdd,1-eq/peak)
    return f"{eq-1:+.0%}/MDD {mdd:.0%}"
for thr in (15.0,25.0):
    info={}
    for c in cl:
        tf=timeframe_to_ms(c.timeframe)
        for i in iso._real_idx(c,4,"full",0.04,thr):
            x=c.pool[4]["full"][i]
            info[(int(x.entry_time),int(x.exit_time))]=("STOP" in str(x.reason).upper(), x.exit_time//tf-x.entry_time//tf)
    for seg in ("is","oos_warm"):
        tr=T[(0.04,4,thr,seg)]; miss=0; base=[];n0=[];n01=[];half=[]
        for e,x,r,w in tr:
            k=info.get((int(e),int(x)))
            if k is None: miss+=1; k=(False,9)
            st,b=k; base.append((x,r))
            if not(st and b==0): n0.append((x,r))
            if not(st and b<=1): n01.append((x,r))
            half.append((x, r*0.5 if (st and b==0) else r))
        print(f"%K<{thr:.0f} {seg:>8}: 지금 {len(base)}거래 {path(base)} | 진입봉 손절 제거 {len(n0)} {path(n0)} | +다음봉 제거 {len(n01)} {path(n01)} | 진입봉 손절 절반 {path(half)} (매칭실패 {miss})")
