import gzip, pickle, __main__, bisect, statistics as st
from backtest import wan424_stoch_isolation as iso, wan424_stoch_ob_arm as arm, harness
from data.storage import OhlcvStore
from data.models import timeframe_to_ms
__main__.IsoCell = iso.IsoCell
arm.register_3h()
with gzip.open("backtest/cache/wan424_iso_work.pkl.gz") as fh: cl=pickle.load(fh)["cells"]
store=OhlcvStore(harness.DB_PATH)
rows=[]  # (arm?, 결과R, 진입봉종가R, 진입봉고가R)
for c in cl:
    tf=timeframe_to_ms(c.timeframe)
    fr=store.load(c.symbol,c.timeframe,start_ms=0,end_ms=10**14)
    if fr.empty: continue
    t=[int(v) for v in fr["open_time"]]; cc=fr["close"].tolist(); hh=fr["high"].tolist()
    armset=set(iso._real_idx(c,4,"full",0.04,15.0))
    for i,x in enumerate(c.pool[4]["full"]):
        e,s=float(x.entry_price),float(x.stop_price)
        if e<=s: continue
        b=(int(x.entry_time)//tf)*tf; j=bisect.bisect_left(t,b)
        if j>=len(t) or t[j]!=b: continue
        R=(float(x.exit_price)-e)/(e-s)
        rows.append((i in armset, arm.stop_width(x)>=0.04, R, (cc[j]-e)/(e-s), (hh[j]-e)/(e-s), "STOP" in str(x.reason).upper()))
def show(name, rs):
    for lab,f in [("큰 수익 (R≥1)",lambda r:r[2]>=1),("수익 (0<R<1)",lambda r:0<r[2]<1),("손실 (손절 아님)",lambda r:r[2]<=0 and not r[5]),("손절",lambda r:r[5])]:
        g=[r for r in rs if f(r)]
        if not g: continue
        cl_=sorted(r[3] for r in g); hi=sorted(r[4] for r in g)
        up=sum(1 for r in g if r[3]>0)/len(g)
        print(f"  {name} {lab:>14}: {len(g):5d}건 · 진입봉 종가 중앙 {st.median(cl_):+.2f}R · 진입봉 고가 중앙 {st.median(hi):+.2f}R · 진입봉이 진입가 위에서 마감 {up:.0%}")
print("R = (가격 − 진입가) ÷ (진입가 − 손절가). 결과R은 ts4 청산 기준(수수료 전).")
show("추천팔", [r for r in rows if r[0]])
show("전체존 4%↑", [r for r in rows if r[1]])
show("전체존", rows)
