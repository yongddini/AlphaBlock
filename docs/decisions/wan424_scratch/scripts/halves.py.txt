import gzip, pickle, __main__, sys, numpy as np
from datetime import datetime, timezone
from backtest import harness
from backtest import wan424_stoch_isolation as iso
from backtest import wan424_matched_null as mn
from backtest import wan424_stoch_ob_arm as arm
from backtest.run import parse_date_ms
from backtest.harness import IS_FRACTION
__main__.IsoCell = iso.IsoCell
floor=float(sys.argv[1]); thr=float(sys.argv[2]); hold=4; DRAWS=200
with gzip.open("backtest/cache/wan424_iso_work.pkl.gz") as fh:
    cells = {(c.symbol,c.timeframe):c for c in pickle.load(fh)["cells"]}
base = arm.build_base_payloads(jobs=1, payload_dir=arm.DEFAULT_PAYLOAD_DIR)
s=parse_date_ms(harness.DEFAULT_START); e=parse_date_ms(harness.DEFAULT_END); b=s+int((e-s)*IS_FRACTION); mid=(s+b)//2
d=lambda ms: datetime.fromtimestamp(ms/1000,tz=timezone.utc).strftime("%Y-%m")
parts={"is 전반":("is",s,mid),"is 후반":("is",mid,b),"oos":("oos_warm",b,e)}
SEGS=("is","oos_warm")
def cut(facts):
    return {k:[f.net_r for f in facts[seg] if lo<=f.entry_time<hi] for k,(seg,lo,hi) in parts.items()}
real=cut(mn.place_facts(iso._select(base,cells,hold=hold,floor=floor,thr=thr,mode="real"),segments=SEGS))
nd=[cut(mn.place_facts(iso._select(base,cells,hold=hold,floor=floor,thr=thr,mode="null",draw=k),segments=SEGS)) for k in range(DRAWS)]
print(f"=== 하한 {floor:.0%} · ts{hold} · %K<{thr:.0f} ===")
for k,(seg,lo,hi) in parts.items():
    rs=real[k]; rm=np.mean(rs); nm=[np.mean(x[k]) for x in nd if x[k]]
    p=np.mean([v>=rm for v in nm]); sig="✅" if (p<=0.05 and rm>np.mean(nm) and len(rs)>=20) else "—"
    print(f"  {k} ({d(lo)}~{d(hi)}): %K {rm:+.3f}({len(rs)}) · 무작위 {np.mean(nm):+.3f} · p {p:.3f} {sig}")
