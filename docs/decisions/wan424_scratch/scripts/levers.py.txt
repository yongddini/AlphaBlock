"""MDD 레버 — 후보를 북에 넣기 전에 규칙으로 거른다(엔진 무변경). 실제 북 · 복리 켬."""
import gzip, pickle, __main__, sys, dataclasses
from backtest import harness
from backtest import wan424_stoch_isolation as iso
from backtest import wan424_stoch_ob_arm as arm
from backtest.book_cli import iter_book_segments
from backtest.leverage_book import LeverageBookParams
from backtest.run import parse_date_ms
__main__.IsoCell = iso.IsoCell
floor=float(sys.argv[1]); thr=float(sys.argv[2]); hold=4
with gzip.open("backtest/cache/wan424_iso_work.pkl.gz") as fh:
    cells = {(c.symbol,c.timeframe):c for c in pickle.load(fh)["cells"]}
base = arm.build_base_payloads(jobs=1, payload_dir=arm.DEFAULT_PAYLOAD_DIR)
real = iso._select(base,cells,hold=hold,floor=floor,thr=thr,mode="real")

def prefilter(items, sym_excl, max_open):
    """items: [(cand, key)] → 진입 시각 순으로 받아들일지. 청산 시각 ≤ 진입이면 먼저 닫는다(북 규약)."""
    if not sym_excl and max_open is None: return set(range(len(items)))
    order = sorted(range(len(items)), key=lambda i: (items[i][0].entry_time, items[i][1]))
    open_=[]; keep=set()
    for i in order:
        c,(sym,tf) = items[i]
        open_=[(x,s) for x,s in open_ if x > c.entry_time]
        if sym_excl and any(s==sym for _,s in open_): continue
        if max_open is not None and len(open_) >= max_open: continue
        open_.append((c.exit_time, sym)); keep.add(i)
    return keep

def payloads_for(seg, sym_excl, max_open):
    raw = "is" if seg=="is" else "full"
    items=[]
    for pi,p in enumerate(real):
        for c in p.candidates.get(raw,()):
            if seg=="oos_warm" and c.trigger_time < p.boundary_ms: continue
            items.append((c,(p.symbol,p.timeframe),pi))
    keep = prefilter([(c,k) for c,k,_ in items], sym_excl, max_open)
    per={pi:[] for pi in range(len(real))}
    for i,(c,k,pi) in enumerate(items):
        if i in keep: per[pi].append(c)
    return [dataclasses.replace(p, candidates={**p.candidates, raw: tuple(per[pi])}) for pi,p in enumerate(real)]

ARMS=[("A0 기준",False,None,None),("A1 종목당 1개",True,None,None),("A2 동시 ≤10",False,10,None),
      ("A3 동시 ≤5",False,5,None),("A4 A1+A2",True,10,None),("A5 하루 −5R 멈춤",False,None,-5.0)]
print(f"=== 하한 {floor:.0%} · ts{hold} · %K<{thr:.0f} === (실제 북 · 복리)")
for name,se,mo,dl in ARMS:
    out=[]
    for seg in ("is","oos_warm"):
        pay=payloads_for(seg,se,mo)
        r=list(iter_book_segments(pay, book=LeverageBookParams(), segments=(seg,),
            start_ms=parse_date_ms(harness.DEFAULT_START), end_ms=parse_date_ms(harness.DEFAULT_END),
            include_reentry=False, compound_sizing=True, min_stop_distance_fraction=0.0,
            take_profit_liquidity=harness.ADOPTED_TAKE_PROFIT_LIQUIDITY, daily_loss_limit_r=dl))[0].row
        ratio = r.total_return/r.max_drawdown if r.max_drawdown>0 else float('nan')
        out.append(f"{seg}: {r.num_trades:4d}거래 복리 {r.total_return*100:+5.0f}% MDD {r.max_drawdown*100:3.0f}% 비율 {ratio:4.2f} 동시최대 {r.peak_concurrency:2d}")
    print(f"{name:<14} | {out[0]} | {out[1]}", flush=True)
