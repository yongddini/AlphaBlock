import gzip, pickle, __main__, collections, datetime as dt
from backtest import wan424_stoch_isolation as iso
from data.models import timeframe_to_ms
__main__.IsoCell = iso.IsoCell
with gzip.open("backtest/cache/wan424_iso_work.pkl.gz") as fh: cl=pickle.load(fh)["cells"]
KST=dt.timezone(dt.timedelta(hours=9)); day=lambda ms:str(dt.datetime.fromtimestamp(ms/1000,KST).date())
EV={"2021-05-19","2021-12-04","2022-01-22","2022-05-10","2022-05-12","2022-06-13","2024-08-05","2025-02-03"}
TFS=("1h","2h","3h","4h","6h","8h","12h","1d","1w")
for thr in (15.0,25.0):
    S={k:collections.defaultdict(lambda:[0,0,0,0.0]) for k in ("폭락일","평소")}  # 진입, 손절, 진입봉/다음봉 손절, net R 근사
    for c in cl:
        tf=timeframe_to_ms(c.timeframe)
        for i in iso._real_idx(c,4,"full",0.04,thr):
            x=c.pool[4]["full"][i]; k="폭락일" if day(x.entry_time) in EV else "평소"; s=S[k][c.timeframe]
            s[0]+=1
            r=(x.exit_price-x.entry_price)/(x.entry_price-x.stop_price); s[3]+=r
            if "STOP" in str(x.reason).upper():
                s[1]+=1
                if x.exit_time//tf - x.entry_time//tf <=1: s[2]+=1
    print(f"\n=== 하한4% · ts4 · %K<{thr:.0f} · 6년 ===")
    print(f"{'TF':>4} | {'폭락일 진입':>9} {'손절%':>6} {'2봉내 깨짐':>9} {'그로스R 평균':>10} | {'평소 진입':>8} {'손절%':>6} {'그로스R 평균':>10}")
    for tf in TFS:
        a=S["폭락일"][tf]; b=S["평소"][tf]
        f=lambda v:(f"{v[1]/v[0]:5.0%}",f"{(v[2]/v[1] if v[1] else 0):8.0%}",f"{v[3]/v[0]:+9.2f}") if v[0] else ("   —","      —","       —")
        fa=f(a); fb=f(b)
        print(f"{tf:>4} | {a[0]:9d} {fa[0]:>6} {fa[1]:>9} {fa[2]:>10} | {b[0]:8d} {fb[0]:>6} {fb[2]:>10}")
