import gzip, pickle, __main__, collections, datetime as dt
from backtest import wan424_stoch_isolation as iso
__main__.IsoCell = iso.IsoCell
with gzip.open("backtest/cache/wan424_iso_work.pkl.gz") as fh:
    cells = pickle.load(fh)["cells"]
KST = dt.timezone(dt.timedelta(hours=9))
c0 = cells[0].pool[4]["full"][0]
print("reasons:", {str(c.pool[4]["full"][i].reason) for c in cells[:5] for i in range(min(20,len(c.pool[4]["full"])))})
for thr in (15.0, 25.0):
    days = collections.defaultdict(list)
    for c in cells:
        for i in iso._real_idx(c, 4, "full", 0.04, thr):
            cand = c.pool[4]["full"][i]
            d = dt.datetime.fromtimestamp(cand.entry_time/1000, KST).date()
            days[d].append((c.symbol, c.timeframe, cand))
    top = sorted(days.items(), key=lambda kv: -len(kv[1]))[:5]
    print(f"\n=== 하한4% · ts4 · %K<{thr:.0f} (전체 {sum(map(len,days.values()))}거래 / {len(days)}일) ===")
    for d, xs in top:
        syms = {s for s,_,_ in xs}; tfs = collections.Counter(t for _,t,_ in xs)
        stops = sum(1 for *_,cand in xs if "STOP" in str(cand.reason).upper())
        rs=[(cand.exit_price-cand.entry_price)/(cand.entry_price-cand.stop_price) for *_,cand in xs]
        print(f"{d}: {len(xs)}건 · {len(syms)}종목 · 손절 {stops} · 그로스R 합 {sum(rs):+.1f} (평균 {sum(rs)/len(rs):+.2f})")
