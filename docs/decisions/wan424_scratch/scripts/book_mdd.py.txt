import gzip, pickle, __main__
from backtest import harness
from backtest import wan424_stoch_isolation as iso
from backtest import wan424_matched_null as mn
from backtest import wan424_stoch_ob_arm as arm
from backtest.book_cli import iter_book_segments
from backtest.leverage_book import LeverageBookParams
from backtest.run import parse_date_ms
__main__.IsoCell = iso.IsoCell
with gzip.open("backtest/cache/wan424_iso_work.pkl.gz") as fh:
    cells = {(c.symbol,c.timeframe):c for c in pickle.load(fh)["cells"]}
base = arm.build_base_payloads(jobs=1, payload_dir=arm.DEFAULT_PAYLOAD_DIR)
def simple(rs):
    cr, cm, dead = arm._equity_path(rs, compound=True)
    return ("파산" if dead else f"{cr*100:+.0f}%"), cm
print("하한 | 구간 | 단순계산 복리/MDD | 실제북 복리/MDD | 최대동시칸 | 최대동시리스크", flush=True)
for floor in (0.0, 0.02, 0.03, 0.04):
    pay = iso._select(base, cells, hold=4, floor=floor, thr=25.0, mode="real")
    facts = mn.place_facts(pay, segments=("full","oos_warm"))
    segs = iter_book_segments(pay, book=LeverageBookParams(), segments=("full","oos_warm"),
        start_ms=parse_date_ms(harness.DEFAULT_START), end_ms=parse_date_ms(harness.DEFAULT_END),
        include_reentry=False, compound_sizing=True, min_stop_distance_fraction=0.0,
        take_profit_liquidity=harness.ADOPTED_TAKE_PROFIT_LIQUIDITY)
    for seg in segs:
        rs=[x.net_r for x in sorted(facts[seg.segment], key=lambda x: x.exit_time)]
        sr, sm = simple(rs)
        r=seg.row
        print(f"{floor:>3.0%} | {seg.segment:>8} | {sr:>7} / {sm*100:3.0f}% | {r.total_return*100:+9.0f}% / {r.max_drawdown*100:3.0f}% | {r.peak_concurrency:>3} | {r.max_concurrent_risk*100:.1f}%", flush=True)
